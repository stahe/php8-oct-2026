declare(strict_types=1);

// ceci est un commentaire
// variable utilisée sans avoir été déclarée

// un affichage écran

// un tableau avec des éléments de type différent
$tableau = ["un", "deux", 3, 4];
// son nombre d'éléments
$n = count($tableau);
// une boucle
for ($i = 0; $i < $n; $i++) {
    print "tableau[$i]=$tableau[$i]\n";
}
// initialisation de 2 variables avec le contenu d'un tableau
[$chaine1, $chaine2] = ["chaine1", "chaine2"];
// concaténation des 2 chaînes
$chaine3 = $chaine1 . $chaine2;
// affichage résultat
print "[$chaine1,$chaine2,$chaine3]\n";
// utilisation fonction
affiche($chaine1);
// le type d'une variable peut être connu

// une fonction peut rendre un résultat

print "res1=$res1\n";
// une fonction peut rendre un tableau de valeurs

print "(res1,res2,res3)=[$res1,$res2,$res3]\n";
// on aurait pu récupérer ces valeurs dans un tableau
$t = f2();
for ($i = 0; $i < count($t); $i++) {
    print "t[$i]=$t[$i]\n";
}
// des tests
for ($i = 0; $i < count($t); $i++) {
    // n'affiche que les chaînes
    if (gettype($t[$i]) === "string") {
        print "t[$i]=$t[$i]\n";
    }
}
// opérateurs de comparaison == et ===
if ("2" == 2) {
    print "avec l'opérateur ==, la chaîne 2 est égale à l'entier 2\n";
} else {
    print "avec l'opérateur ==, la chaîne 2 n'est pas égale à l'entier 2\n";
}
if ("2" === 2) {
    print "avec l'opérateur ===, la chaîne 2 est égale à l'entier 2\n";
} else {
    print "avec l'opérateur ===, la chaîne 2 n'est pas égale à l'entier 2\n";
}
// une comparaison dont le résultat a changé avec PHP 8
if (0 == "abc") {
    print "avec l'opérateur ==, l'entier 0 est égal à la chaîne abc\n";
} else {
    print "avec l'opérateur ==, l'entier 0 n'est pas égal à la chaîne abc\n";
}
// d'autres tests
for ($i = 0; $i < count($t); $i++) {
    // n'affiche que les entiers > 10
    if (is_int($t[$i]) && $t[$i] > 10) {
        print "t[$i]=$t[$i]\n";
    }
}
// une boucle while
$t = [8, 5, 0, -2, 3, 4];
$i = 0;
$somme = 0;
while ($i < count($t) && $t[$i] > 0) {
    print "t[$i]=$t[$i]\n";
    $somme += $t[$i];   // $somme = $somme + $t[$i]
    $i++;               // $i = $i + 1
}
print "somme=$somme\n";

// fin programme
exit;

{
    // affiche $chaine
    print "chaine=$chaine\n";
}

{
    // affiche le type de $variable
    print "type[variable $" . $name . "]=" . gettype($variable) . "\n";
}

//----------------------------------
function f1(int $param): int
{
    // ajoute 10 à $param
    return $param + 10;
}

//----------------------------------
function f2(): array
{
    // rend 3 valeurs
    return ["un", 0, 100];

    Commentaires :

    • ligne 4 : demande à PHP de vérifier strictement les types déclarés des paramètres et des résultats des fonctions. Nous y reviendrons au paragraphe consacré aux changements de types. Pour l'instant, retenons que tous les scripts du cours commencent ainsi ;
    • ligne 8 : en PHP, on ne déclare pas les variables. Celles-ci ont un type dynamique qui peut varier au cours du temps. [$nom] représente la variable d'identifiant [nom] ;
    • ligne 10 : pour écrire à l'écran, on peut utiliser l'instruction [print] ou l'instruction [echo] ;
    • ligne 12 : la notation [[...]] permet de définir un tableau. La notation [$tableau[$i]] représente l'élément n° [$i] du tableau [$tableau], le premier élément ayant le n° 0. La notation plus ancienne [array("un", "deux", 3, 4)] est toujours acceptée, mais nous ne l'utiliserons pas ;
    • ligne 14 : la fonction [count($tableau)] rend le nombre d'éléments du tableau [$tableau] ;
    • lignes 16-18 : une boucle. Celle-ci n'ayant qu'une instruction, les accolades sont facultatives. Dans la suite de ce document, nous mettrons systématiquement les accolades quel que soit le nombre d'instructions ;
    • ligne 17 : les chaînes de caractères sont entourées de guillemets " ou d'apostrophes '. À l'intérieur de guillemets, les variables [$variable] sont évaluées (on dit aussi « interpolées »), mais pas à l'intérieur d'apostrophes ;
    • ligne 20 : on peut affecter les éléments d'un tableau à plusieurs variables en une seule opération. Ici [$chaine1="chaine1"] et [$chaine2="chaine2"]. Cette écriture, appelée déstructuration, remplace la fonction [list] des anciennes versions de PHP : [list($chaine1, $chaine2) = ...] est toujours accepté ;
    • ligne 22 : l'opérateur . (point) est l'opérateur de concaténation de chaînes ;
    • lignes 91-95 : le mot clé [function] définit une fonction. Une fonction peut être définie n'importe où dans le script, même après son utilisation. Ici, [string $chaine] indique que le paramètre [$chaine] doit être une chaîne de caractères et [: void] que la fonction ne rend aucun résultat. Ces déclarations de types sont facultatives en PHP, mais nous les utiliserons systématiquement : elles documentent le code et permettent à PHP (et à l'éditeur) de détecter des erreurs ;
    • ligne 98 : le type [mixed] (PHP 8.0) signifie « n'importe quel type ». Le paramètre [$variable] peut recevoir un entier, une chaîne, un tableau… ;
    • ligne 101 : la fonction prédéfinie [gettype($variable)] rend une chaîne de caractères représentant le type de [$variable]. Les résultats montrent que ce type peut changer au cours du temps : [$n] est d'abord un entier (integer) puis une chaîne (string) ;
    • ligne 105 : une fonction rend un résultat par l'instruction [return]. Le type de ce résultat est déclaré après les parenthèses : [: int]. Le code appelant peut ignorer ou récupérer les résultats d'une fonction ;
    • ligne 112 : une fonction peut rendre plusieurs valeurs en les rassemblant dans un tableau ;
    • lignes 53-62 : l'opérateur [===] compare deux éléments de façon stricte : il faut qu'ils aient le même type et la même valeur pour être égaux. L'opérateur [] est moins strict : deux éléments peuvent être égaux sans être du même type. Dans ce cas, la comparaison se fait après conversion des deux éléments dans un même type. Il est assez facile « d'oublier » la présence de ces conversions implicites et d'aboutir ainsi à des résultats imprévus, tels que de découvrir qu'une condition est vraie alors que vous l'attendiez fausse. Pour éviter cet écueil, nous utiliserons systématiquement l'opérateur de comparaison [=] (et son contraire [!==]) ;
    • lignes 64-68 : un exemple de ces conversions implicites. En PHP 7, la chaîne "abc" était convertie en le nombre 0, et l'expression [0 == "abc"] était vraie ! Depuis PHP 8.0, lorsqu'on compare un nombre à une chaîne qui ne représente pas un nombre, c'est le nombre qui est converti en chaîne, et l'expression est fausse. Le comportement de [] a donc changé d'une version à l'autre : raison de plus pour lui préférer [=] ;
    • ligne 72 : la fonction [is_int] rend [true] si son paramètre est un entier. Il existe de même [is_string], [is_float], [is_bool], [is_array]… L'opérateur [&&] est le ET logique. On peut également utiliser les opérateurs logiques [||] (OU) et [!] (NON). PHP connaît aussi les mots clés [and] et [or], mais leur priorité, plus faible que celle de l'affectation, réserve des surprises : nous ne les utiliserons pas ;
    • ligne 80 : une boucle [while]. Elle s'arrête dès que la condition est fausse, ici dès que le tableau est parcouru ou qu'on rencontre un élément négatif ou nul ;
    • ligne 88 : l'instruction [exit] arrête l'exécution du script. Les fonctions qui suivent ne sont pas exécutées à cet endroit mais elles sont connues de tout le script .

    On remarquera que le script ne se termine pas par la balise [?>] de fin de code PHP. Elle n'est pas indispensable. De plus, dans un contexte web, elle peut poser problème si elle est suivie par des espaces ou des marques de fin de ligne, difficiles à détecter car non visibles dans un éditeur de texte. La recommandation [PSR-12] demande d'ailleurs de l'omettre dans les fichiers ne contenant que du PHP. Aussi, dans la suite du document, nous l'omettrons systématiquement.

    Note : dans ce document, nous utiliserons le mot clé [print] pour afficher du texte sur la console. Une autre méthode pour faire la même chose est d'utiliser le mot clé [echo]. Il y a de subtiles différences entre ces deux mots clés, mais dans le contexte de ce document il n'y en aura aucune. Si donc vous préférez utiliser [echo], alors faites-le.

    3.3.2. Usage de VS Code

    Pour travailler sur les scripts du cours, on ouvre le dossier [php8-oct-2026] dans [VS Code] (menu [Fichier > Ouvrir le dossier]). L'arborescence des fichiers apparaît dans l'explorateur, à gauche de la fenêtre. Pour exécuter un script :

    • on ouvre un terminal intégré (menu [Terminal > Nouveau terminal]). Il s'ouvre dans le dossier [php8-oct-2026] ;
    • on se place dans le dossier du script : [cd scripts-console/bases] ;
    • on lance le script : [php bases-01.php]. Les résultats s'affichent dans le terminal. On peut aussi, dans l'explorateur, cliquer droit sur le dossier [bases] et choisir [Ouvrir dans le terminal intégré] .

    Si la commande [php] n'est pas reconnue, c'est que le dossier de PHP n'est pas dans le PATH de Windows : [Laragon] l'y ajoute lorsqu'on coche l'option correspondante de son menu [Outils > Path]. La commande [php -v] doit afficher la version 8.5.

    L'extension [PHP Intelephense] analyse le code au fur et à mesure de la frappe et signale les problèmes par un trait ondulé sous le code fautif. Le survol de ce trait avec la souris affiche le message. Tous les problèmes du dossier sont rassemblés dans le panneau [Problèmes] (menu [Affichage > Problèmes]). Parmi les avertissements utiles :

    • [Undefined variable] : une variable est utilisée sans avoir été initialisée. C'est souvent une faute de frappe sur le nom d'une variable ;
    • [Undefined function] : appel d'une fonction qui n'existe pas, là encore souvent une faute de frappe ;
    • [Expected type 'int'. Found 'string'.] : un paramètre effectif n'a pas le type déclaré par la fonction. C'est l'un des grands bénéfices des déclarations de types : l'erreur est signalée avant même l'exécution ;
    • [Expected 2 arguments. Found 1.] : le nombre de paramètres effectifs ne correspond pas à la définition de la fonction .

    Le survol d'un nom de fonction affiche sa signature (paramètres, types, résultat) et sa documentation, y compris pour les fonctions prédéfinies de PHP. Pour qu'[Intelephense] connaisse les nouveautés récentes du langage, on vérifie dans les paramètres de [VS Code] (menu [Fichier > Préférences > Paramètres], rechercher « intelephense php version ») que la version de PHP indiquée est la 8.5. En phase d'apprentissage du langage, il est conseillé de corriger tous les avertissements signalés. On apprend ainsi beaucoup de choses.

    [Intelephense] sait aussi mettre en forme un fichier (clic droit, [Mettre en forme le document]) selon les recommandations de codage [PSR-12]. Les PSR (PHP Standard Recommendations) sont des recommandations pour produire un code standard et ainsi faciliter l'interopérabilité et la maintenance de codes écrits par différentes personnes. Elles sont publiées par le [PHP-FIG] sur le site www.php-fig.org. Les recommandations [PSR-1] et [PSR-12] fixent les règles d'écriture du code. Voici celles de [PSR-1] :

    Élément
    Règle
    Constantes de classe
    Les constantes de classe DOIVENT être écrites en majuscules avec le caractère _ comme séparateur.
    Ex : [const TAUX_TVA]
    Méthodes
    Les noms de méthodes DOIVENT être écrits en camelCase.
    Ex : [public function executeBatchImpots()]
    Propriétés
    Les noms de propriétés DEVRAIENT être écrits de façon cohérente dans un même ensemble de code, en StudlyCaps, camelCase ou under_score.
    Ex : [$SalaireAnnuel], [$salaireAnnuel], [$salaire_annuel]
    Effets de bord
    Un fichier DEVRAIT soit déclarer de nouveaux symboles (classes, fonctions, constantes…) sans autre effet de bord, soit exécuter du code avec des effets de bord, mais NE DEVRAIT PAS faire les deux.
    Types
    Les noms de types (classes, interfaces…) DOIVENT être écrits en StudlyCaps. Chaque type est dans un fichier à lui et dans un espace de noms d'au moins un niveau.
    Ex : [class EtudiantBoursier]

    La règle des effets de bord dit que dans un fichier PHP, on doit trouver :

    • soit la déclaration de types (classes, interfaces), de fonctions ou de constantes ;
    • soit du code exécutable sans déclaration de nouveaux types .

    Par facilité, les exemples de ce chapitre ne la respectent pas : chaque script est un fichier unique qui définit ses fonctions et les utilise. Répartir le code dans plusieurs fichiers serait trop lourd pour des exemples basiques. Pour l'application présentée comme fil rouge de ce document, on cherchera, dès qu'elle sera écrite avec des classes, à observer au mieux ces recommandations.

    • ligne 13 : le modèle /(\d+)/ est comparé à la chaîne "12 34". On voit que les sous-chaînes 12 et 34 correspondent au modèle. La comparaison s'arrête à la première sous-chaîne correspondant au modèle. On aura donc [$champs[0]=12] et [$champs[1]=12] ;

    Explicitons les modèles utilisés dans la suite du code :

    1
    $modèle = "/^(.*?)(\d+)(.*?)$/";

    correspond à début de chaîne (^), puis 0 ou plusieurs (*) caractères quelconques (.) puis 1 ou plusieurs (+) chiffres, puis de nouveau 0 ou plusieurs (*) caractères quelconques (.). Le modèle (.*) désigne 0 ou plusieurs caractères quelconques. Un tel modèle va correspondre à n'importe quelle chaîne. Ainsi, dans le modèle /^(.*)(\d+)(.*)$/, le premier sous-modèle (.*) absorberait tous les chiffres sauf le dernier, le seul laissé au sous-modèle (\d+). Le modèle (.*?)(\d+) désigne lui 0 ou plusieurs caractères quelconques jusqu'au sous-modèle suivant (?), ici \d+. Donc les chiffres ne sont maintenant plus absorbés par le modèle (.*). Le modèle ci-dessus correspond donc à [début de chaîne (^), une suite de caractères quelconques (.*?), une suite d'un ou plusieurs chiffres (\d+), une suite de caractères quelconques (.*?), la fin de la chaîne ($)].

    1
    $modèle = "/^\s*(\d\d)\/(\d\d)\/(\d\d)\s*$/";

    correspond à [début de chaîne (^), une suite de 0 ou plusieurs espaces (\s*), 2 chiffres (\d\d), le caractère / (\/), 2 chiffres, /, 2 chiffres, une suite de 0 ou plusieurs espaces (\s*), la fin de chaîne ($)].

    1
    $modèle = "/^\s*([+-]?)\s*(\d+\.\d*|\.\d+|\d+)\s*/";

    $ligne = cutNewLineChar($ligne);

    $infos = explode(":", $ligne);
    // on récupère le login
    $login = array_shift($infos);

    array_shift($infos);

// --------------------------------------------------------------------------
function afficheInfos(array $dico, string $clé): void
{
    // affiche la valeur associée à clé dans le dictionnaire $dico si elle existe
    if (isset($dico[$clé])) {
        // valeur existe - est-ce un tableau ?
        $valeur = $dico[$clé];
        if (is_array($valeur)) {
            print "[$clé," . implode(":", $valeur) . "]\n";
        } else {
            // $valeur n'est pas un tableau
            print "[$clé,$valeur]\n";
        }
    } else {
        // $clé n'est pas une clé du dictionnaire $dico
        print "la clé [$clé] n'existe pas\n";
    }
}

// --------------------------------------------------------------------------
function cutNewLineChar(string $ligne): string
{
    // on supprime la marque de fin de ligne de $ligne si elle existe
    // (\n sous Unix, \r\n sous Windows)
    return rtrim($ligne, "\r\n");
}
